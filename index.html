/** * 修正ポイント: * 1. LockServiceによる同時書き込み防止 * 2. シート取得の効率化 */ function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); // 紐付いたシートを開く const action = e.parameter.action; const excludeSheets = ["完了ログ", "スタッフ名簿"]; // ランキング取得処理 if (action === 'getRanking') { const logSheet = ss.getSheetByName("完了ログ"); const logs = logSheet.getDataRange().getValues(); const rankingMap = {}; for (let i = 1; i < logs.length; i++) { const point = parseInt(logs[i][3]) || 0; const staffName = logs[i][4]; if (!rankingMap[staffName]) { rankingMap[staffName] = 0; } rankingMap[staffName] += point; } const rankingArray = Object.keys(rankingMap).map(name => { return { name: name, totalPoints: rankingMap[name] }; }); return createJsonResponse(rankingArray); } // タスク一覧とスタッフ名簿の取得 const tasks = []; const sheets = ss.getSheets(); sheets.forEach(sheet => { const name = sheet.getName(); if (!excludeSheets.includes(name)) { const vals = sheet.getDataRange().getValues(); for (let i = 1; i < vals.length; i++) { if (vals[i][1]) { let diff = vals[i][2]; let pt = (diff == 1) ? 10 : (diff == 2) ? 20 : (diff == 3) ? 30 : (diff == 4) ? 50 : 5; tasks.push({ category: name, name: vals[i][1], level: "難易度" + diff, point: pt }); } } } }); const staffList = []; const staffSheet = ss.getSheetByName("スタッフ名簿"); if (staffSheet) { const staffData = staffSheet.getDataRange().getValues(); for (let i = 1; i < staffData.length; i++) { let pts = parseInt(staffData[i][1] || 0); let lv = Math.floor(pts / 100) + 1; let title = getTitle(lv); staffList.push({ name: staffData[i][0], points: pts, level: lv, title: title }); } } return createJsonResponse({tasks: tasks, staffList: staffList}); } function doPost(e) { const lock = LockService.getScriptLock(); try { // 30秒間他の書き込みを待機させる（重要！） lock.waitLock(30000); const ss = SpreadsheetApp.getActiveSpreadsheet(); const logSheet = ss.getSheetByName("完了ログ"); const staffSheet = ss.getSheetByName("スタッフ名簿"); const params = JSON.parse(e.postData.contents); // ログ追記 logSheet.appendRow([new Date(), params.category, params.taskName, params.point, params.staffName]); // ポイント更新 if (staffSheet) { const staffData = staffSheet.getDataRange().getValues(); for (let i = 1; i < staffData.length; i++) { if (staffData[i][0] === params.staffName) { const currentPoint = parseInt(staffData[i][1] || 0); staffSheet.getRange(i + 1, 2).setValue(currentPoint + parseInt(params.point)); break; } } } return createJsonResponse({status: "success"}); } catch(err) { return createJsonResponse({status: "error", message: err.message}); } finally { lock.releaseLock(); } } // 共通関数: 称号の判定 function getTitle(level) { if (level >= 30) return "店舗守護神"; if (level >= 20) return "清掃神"; if (level >= 10) return "美化マスター"; if (level >= 5) return "サロンキーパー"; return "見習いクリーン"; } // 共通関数: JSONレスポンス作成 function createJsonResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON); }
